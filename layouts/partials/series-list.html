    2. {{- else if eq $gi $cur -}}
  2. {{ $e.title }}
    3. {{- else -}}
  3. {{ $e.title }}
    4. {{- end -}} {{- end -}}
{{- end -}}

  1. {{ $e.title }}{{ i18n "seriesUnwritten" | default "unwritten" }}

  2. {{ $e.title }}
